graph.pyPython · LangGraph 1.x
from langgraph.graph import StateGraph, START, END
from langgraph.types import interrupt
from .state import FeatureState
from .steps import plan, write_tests, implement, review, fix, gate, open_pr, summary

MAX_FIX_LOOPS = 3

def approve(stage: str):
    # human-in-the-loop: pauses the run until Approve/Reject in VS Code
    def node(state: FeatureState):
        decision = interrupt({"stage": stage, "run_id": state["run_id"]})
        return {"approvals": [*state["approvals"], decision]}
    return node

def after_check(passed_to: str):
    def route(state: FeatureState) -> str:
        if state["last_check_passed"]:
            return passed_to
        return "fix" if state["fix_loops"] < MAX_FIX_LOOPS else "summary"
    return route

g = StateGraph(FeatureState)
g.add_node("plan", plan)
g.add_node("approve_plan", approve("plan"))
g.add_node("write_tests", write_tests)   # separate agent, sees the spec only
g.add_node("implement", implement)
g.add_node("review", review)             # independent agent session
g.add_node("fix", fix)
g.add_node("gate", gate)                 # tests + invariant property tests
g.add_node("approve_pr", approve("pr"))
g.add_node("open_pr", open_pr)
g.add_node("summary", summary)

g.add_edge(START, "plan")
g.add_edge("plan", "approve_plan")
g.add_edge("approve_plan", "write_tests")
g.add_edge("write_tests", "implement")
g.add_edge("implement", "review")
g.add_conditional_edges("review", after_check("gate"))
g.add_conditional_edges("gate", after_check("approve_pr"))
g.add_conditional_edges("fix", lambda s: s["failed_check"])  # back to review or gate
g.add_edge("approve_pr", "open_pr")
g.add_edge("open_pr", "summary")
g.add_edge("summary", END)

graph = g.compile()

Rules that make it trustworthy
• A release is immutable; its digest is checked before install.
• A running task never switches to a new version.
• A new version is offered, not auto-run, unless the team sets that mode.
• Agent permissions from workflow.toml are shown before install.
• The hub never executes workflow code and never sees CLI credentials.
